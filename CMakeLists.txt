cmake_minimum_required(VERSION 3.16)
project(kinectcam VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "kinectcam requires Linux V4L2")
endif()
find_package(freenect2 REQUIRED CONFIG)
find_package(Threads REQUIRED)

add_library(kinectcam_pixels src/pixels.cpp)
target_include_directories(kinectcam_pixels PUBLIC src)
add_library(kinectcam_v4l2 src/v4l2_output.cpp)
target_include_directories(kinectcam_v4l2 PUBLIC src)
add_executable(kinectcam main.cpp src/options.cpp)
target_include_directories(kinectcam PRIVATE ${freenect2_INCLUDE_DIRS})
target_link_libraries(kinectcam PRIVATE kinectcam_pixels kinectcam_v4l2 ${freenect2_LIBRARIES} Threads::Threads)
foreach(target kinectcam kinectcam_pixels kinectcam_v4l2)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()

include(CTest)
if(BUILD_TESTING)
    add_executable(test_pixels tests/test_pixels.cpp)
    target_link_libraries(test_pixels PRIVATE kinectcam_pixels)
    add_test(NAME pixels COMMAND test_pixels)
    add_executable(test_v4l2 tests/test_v4l2.cpp)
    target_link_libraries(test_v4l2 PRIVATE kinectcam_v4l2)
    target_link_options(test_v4l2 PRIVATE -Wl,--wrap=open,--wrap=close,--wrap=fstat,--wrap=ioctl,--wrap=write,--wrap=poll)
    add_test(NAME v4l2_boundary COMMAND test_v4l2)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:kinectcam>)
    add_executable(kinectcam_test_io main.cpp src/options.cpp tests/fake_video.cpp)
    target_include_directories(kinectcam_test_io PRIVATE ${freenect2_INCLUDE_DIRS})
    target_link_libraries(kinectcam_test_io PRIVATE kinectcam_pixels kinectcam_v4l2 ${freenect2_LIBRARIES} Threads::Threads)
    target_compile_options(kinectcam_test_io PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(kinectcam_test_io PRIVATE -Wl,--wrap=open,--wrap=close,--wrap=fstat,--wrap=ioctl)
    add_test(NAME synthetic_streams COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_streams.py $<TARGET_FILE:kinectcam_test_io>)
    add_test(NAME setup COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_setup.py ${CMAKE_CURRENT_SOURCE_DIR}/setup-loopback.sh)
    find_program(SSH_KEYGEN_EXECUTABLE ssh-keygen)
    find_program(GIT_EXECUTABLE git)
    if(SSH_KEYGEN_EXECUTABLE AND GIT_EXECUTABLE)
        add_test(NAME aur_publish COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_publish_aur.py)
    endif()
endif()

include(GNUInstallDirs)
configure_file(packaging/kinectcam.service.in kinectcam.service @ONLY)
install(TARGETS kinectcam RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS setup-loopback.sh DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/kinectcam)
install(FILES packaging/99-kinectcam.rules DESTINATION lib/udev/rules.d)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/kinectcam.service DESTINATION lib/systemd/system)
install(FILES packaging/kinectcam.conf.example DESTINATION ${CMAKE_INSTALL_DATADIR}/kinectcam)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES docs/aur-publishing.md docs/validation.md DESTINATION ${CMAKE_INSTALL_DOCDIR}/docs)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/kinectcam)
